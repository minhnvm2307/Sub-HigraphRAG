\documentclass[tikz,border=4pt]{standalone}
\usepackage{amsmath,amssymb}
\usetikzlibrary{arrows.meta,positioning,fit,backgrounds,calc}

\begin{document}
\begin{tikzpicture}[
  font=\small,
  box/.style   = {draw, rounded corners=2pt, align=center, inner sep=4pt, minimum height=7mm},
  learn/.style = {box, fill=blue!8,   draw=blue!55,   thick},
  fixed/.style = {box, fill=gray!8,   draw=gray!60},
  data/.style  = {box, fill=green!8,  draw=green!50!black},
  out/.style   = {box, fill=orange!10,draw=orange!70!black},
  grp/.style   = {draw, dashed, rounded corners=4pt, inner sep=7pt, draw=black!35},
  lbl/.style   = {font=\scriptsize\itshape, text=black!65},
  ar/.style    = {-{Latex[length=2mm]}, thick, draw=black!65},
]

% ---------------------------------------------------------------- inputs
\node[data] (q)   {question $q \to \mathbf{z}_q$};
\node[data, right=5mm of q]  (g) {subgraph $\mathcal{G}_q$, anchors $\mathcal{A}$, $\mathcal{V}_{\mathrm{txt}}$};
\node[data, right=5mm of g]  (p) {$\mathrm{lex}(q,r)$, $\cos_{\mathrm{BGE}}(q,r)$};

% ---------------------------------------------------------------- M1
\node[fixed, below=9mm of q.south west, anchor=north west, text width=34mm]
  (feat) {$\mathbf{u}(q,r)=[\mathbf{z}_q;\mathbf{z}_r;\mathbf{z}_q\!\odot\!\mathbf{z}_r;\mathrm{lex};\cos;\boldsymbol{\rho}]$};
\node[learn, right=6mm of feat, text width=26mm]
  (pi)   {$\pi(r,d\,|\,q)=\sigma(\mathrm{MLP}_\pi(\mathbf{u})_d)$\\[1pt]{\scriptsize one-hop relevance factor}};
\node[learn, right=5mm of pi, text width=26mm]
  (phi)  {$\varphi(r,d\,|\,q)=\sigma(\mathrm{MLP}_\varphi(\mathbf{u})_d)$\\[1pt]{\scriptsize stop factor}};

\begin{scope}[on background layer]
  \node[grp, fill=blue!3, fit=(feat)(pi)(phi)] (M1) {};
\end{scope}
\node[lbl, anchor=north west] at ($(M1.north west)+(1mm,-0.5mm)$)
  {M1 -- the only learned parameters; inputs depend on $(q,r)$ only};

% ---------------------------------------------------------------- M2 / M3
\node[fixed, below=12mm of M1.south west, anchor=north west, text width=92mm]
  (bf) {$\displaystyle \log p^{(t+1)}_a(v)=\max\Big(\log p^{(t)}_a(v),\;
        \max_{(u,r,v)\in\mathcal{E}^{\pm}}\big[\log p^{(t)}_a(u)+\log\pi(r,d\,|\,q)\big]\Big)$};
\node[fixed, below=3.5mm of bf.south west, anchor=north west, text width=92mm]
  (nb) {non-backtracking: forbid $(r,\bar d)$ right after $(r,d)$ \;$\Rightarrow$\;
        per-node Viterbi top-2 state $(b_v,\beta_v,s_v,\varsigma_v)$ \;--\; 4 scalars, no vectors};
\begin{scope}[on background layer]
  \node[grp, fill=gray!3, fit=(bf)(nb)] (M2) {};
\end{scope}
\node[lbl, anchor=north west] at ($(M2.north west)+(1mm,-0.5mm)$)
  {M2--M3 -- $(\max,+)$ Bellman--Ford, $T$ steps, $O(A\,T\,|\mathcal{E}|)$};

% ---------------------------------------------------------------- M4
\node[fixed, below=12mm of M2.south west, anchor=north west, text width=92mm]
  (mm) {$\Psi(v,\tau)=\max\big(\underbrace{\mathrm{stop}(v,\tau)}_{\log\varphi \text{ if } v\in\mathcal{V}_{\mathrm{txt}}},\;
        \max_{(r,d)\neq\bar\tau}[\log\pi(r,d\,|\,q)+\Psi(w,(r,d))]\big)$,\quad $\le L$ steps};
\node[fixed, below=3.5mm of mm.south west, anchor=north west, text width=92mm]
  (sc) {$\mathrm{score}_a(e)=\max_d\big[\mathrm{pre}^{d}_a(e)+\Psi(\mathrm{far}^d(e),(r_e,d))\big]$
        \;=\; best \emph{complete} walk through $e$ ending at a text entity};
\begin{scope}[on background layer]
  \node[grp, fill=gray!3, fit=(mm)(sc)] (M4) {};
\end{scope}
\node[lbl, anchor=north west] at ($(M4.north west)+(1mm,-0.5mm)$)
  {M4 -- complete-path max-marginal (test time); no extra parameters};

% ---------------------------------------------------------------- M5
\node[fixed, below=12mm of M4.south west, anchor=north west, text width=44mm]
  (un) {$\mathrm{union}(e)=\max_a \mathrm{score}_a(e)$};
\node[fixed, right=4mm of un, text width=44mm]
  (it) {$\mathrm{inter}(e)=\max_{a,d}\big[\mathrm{score}^d_a(e)
        +\!\!\sum_{a'\neq a}\!\log p_{a'}(\mathrm{far}^d(e))\big]$};
\node[fixed, below=3.5mm of un.south west, anchor=north west, text width=92mm]
  (fu) {$s(e)=\operatorname{logaddexp}(\mathrm{union},\mathrm{inter})$\quad
        {\scriptsize($A\!=\!1 \Rightarrow s(e)=\mathrm{union}(e)+\log 2$: exactly the single-anchor model)}};
\begin{scope}[on background layer]
  \node[grp, fill=gray!3, fit=(un)(it)(fu)] (M5) {};
\end{scope}
\node[lbl, anchor=north west] at ($(M5.north west)+(1mm,-0.5mm)$)
  {M5 -- multi-anchor fusion: max (union) $\cup$ product t-norm (intersection), parameter-free};

% ---------------------------------------------------------------- M6 / output
\node[out, below=11mm of M5.south west, anchor=north west, text width=44mm]
  (lg) {$\ell(e)=\operatorname{logit}\!\big(e^{\,s(e)-\log 2}\big)$\\[1pt]
        $\mathcal{L}=\mathrm{BCE}(\ell,y)+\mathcal{L}_{\mathrm{stop}}$};
\node[out, right=4mm of lg, text width=44mm]
  (tk) {top-$k$ triples ($k\!=\!100$)\\[1pt]$\to$ path-style prompt $\to$ LLM reasoner};

% ---------------------------------------------------------------- arrows
\draw[ar] (q.south)   -- ++(0,-3mm) -| (feat.north);
\draw[ar] (p.south)   -- ++(0,-3mm) -| ([xshift=6mm]feat.north east);
\draw[ar] (feat.east) -- (pi.west);
\draw[ar] (pi.east)   -- (phi.west);
\draw[ar] (pi.south)  .. controls +(0,-6mm) and +(0,6mm) .. ([xshift=-10mm]bf.north);
\draw[ar] (g.east)    -- ++(5mm,0) |- (bf.east);
\draw[ar] (M2.south)  -- (M4.north);
\draw[ar] (phi.south) .. controls +(22mm,-26mm) and +(14mm,16mm) .. (mm.east);
\draw[ar] (M4.south)  -- (M5.north);
\draw[ar] (M5.south)  -- ++(0,-4mm) -| (lg.north);
\draw[ar] (lg.east)   -- (tk.west);

\end{tikzpicture}
\end{document}
